def get_knowledge_point(name: str) -> dict:
    """
    获取单个知识点详情

    Returns:
        {
            'name': str,
            'category': str,
            'description': str,
            'difficulty': str,  # beginner/intermediate/advanced
            'importance': str,  # low/medium/high/critical
            'estimated_duration': int,  # 分钟
            'tags': List[str],
            'content': str,
            'prerequisites': List[str],  # 前置依赖的知识点名称
            'relations': List[str],  # 关联的知识点名称
            'created_at': str,
            'updated_at': str
        }
    """
    with driver.session() as session:
        result = session.run("""
            MATCH (kp:KnowledgePoint {name: $name})
            OPTIONAL MATCH (kp)<-[:REQUIRES]-(dependent:KnowledgePoint)
            OPTIONAL MATCH (kp)-[:RELATED_TO|SIMILAR_TO]-(related:KnowledgePoint)
            RETURN kp,
                   collect(DISTINCT dependent.name) as prerequisites,
                   collect(DISTINCT related.name) as relations
        """, name=name)

        record = result.single()
        if not record:
            raise GraphEntityNotFoundError(f"Knowledge point '{name}' not found")

        kp = record['kp']
        return {
            'name': kp['name'],
            'category': kp.get('category'),
            'description': kp.get('description'),
            'difficulty': kp.get('difficulty', 'beginner'),
            'importance': kp.get('importance', 'medium'),
            'estimated_duration': kp.get('estimated_duration'),
            'tags': kp.get('tags', []),
            'content': kp.get('content'),
            'prerequisites': [p for p in record['prerequisites'] if p],
            'relations': [r for r in record['relations'] if r],
            'created_at': kp.get('created_at'),
            'updated_at': kp.get('updated_at')
        }


def list_knowledge_points_enhanced(
    category: str = None,
    difficulty: str = None,
    search: str = None
) -> List[dict]:
    """
    获取知识点列表（支持过滤）

    Args:
        category: 分类过滤
        difficulty: 难度过滤
        search: 搜索关键词

    Returns:
        List of knowledge points with basic info
    """
    with driver.session() as session:
        # 构建查询条件
        conditions = []
        params = {}

        if category:
            conditions.append("kp.category = $category")
            params['category'] = category

        if difficulty:
            conditions.append("kp.difficulty = $difficulty")
            params['difficulty'] = difficulty

        if search:
            conditions.append("kp.name CONTAINS $search OR kp.description CONTAINS $search")
            params['search'] = search

        where_clause = "WHERE " + " AND ".join(conditions) if conditions else ""

        query = f"""
            MATCH (kp:KnowledgePoint)
            {where_clause}
            OPTIONAL MATCH (kp)<-[:REQUIRES]-(dependent:KnowledgePoint)
            OPTIONAL MATCH (kp)-[:RELATED_TO|SIMILAR_TO]-(related:KnowledgePoint)
            RETURN kp,
                   count(DISTINCT dependent) as prerequisite_count,
                   count(DISTINCT related) as relation_count
            ORDER BY kp.name
        """

        result = session.run(query, **params)

        points = []
        for record in result:
            kp = record['kp']
            points.append({
                'name': kp['name'],
                'category': kp.get('category'),
                'description': kp.get('description'),
                'difficulty': kp.get('difficulty', 'beginner'),
                'importance': kp.get('importance', 'medium'),
                'estimated_duration': kp.get('estimated_duration'),
                'tags': kp.get('tags', []),
                'prerequisite_count': record['prerequisite_count'],
                'relation_count': record['relation_count']
            })

        return points


def create_knowledge_point(data: dict) -> dict:
    """
    创建新知识点

    Args:
        data: {
            'name': str (required),
            'category': str (optional),
            'description': str (optional),
            'difficulty': str (optional),
            'importance': str (optional),
            'estimated_duration': int (optional),
            'tags': List[str] (optional),
            'content': str (optional)
        }

    Returns:
        Created knowledge point
    """
    import datetime

    if not data.get('name'):
        raise ValueError("Knowledge point name is required")

    with driver.session() as session:
        # 检查是否已存在
        existing = session.run(
            "MATCH (kp:KnowledgePoint {name: $name}) RETURN kp",
            name=data['name']
        ).single()

        if existing:
            raise ValueError(f"Knowledge point '{data['name']}' already exists")

        # 创建节点
        now = datetime.datetime.utcnow().isoformat()
        result = session.run("""
            CREATE (kp:KnowledgePoint {
                name: $name,
                category: $category,
                description: $description,
                difficulty: $difficulty,
                importance: $importance,
                estimated_duration: $estimated_duration,
                tags: $tags,
                content: $content,
                created_at: $created_at,
                updated_at: $updated_at
            })
            RETURN kp
        """,
            name=data['name'],
            category=data.get('category'),
            description=data.get('description'),
            difficulty=data.get('difficulty', 'beginner'),
            importance=data.get('importance', 'medium'),
            estimated_duration=data.get('estimated_duration'),
            tags=data.get('tags', []),
            content=data.get('content'),
            created_at=now,
            updated_at=now
        )

        return get_knowledge_point(data['name'])


def update_knowledge_point(name: str, data: dict) -> dict:
    """
    更新知识点

    Args:
        name: 原知识点名称
        data: 更新数据（同create_knowledge_point）

    Returns:
        Updated knowledge point
    """
    import datetime

    with driver.session() as session:
        # 检查是否存在
        existing = session.run(
            "MATCH (kp:KnowledgePoint {name: $name}) RETURN kp",
            name=name
        ).single()

        if not existing:
            raise GraphEntityNotFoundError(f"Knowledge point '{name}' not found")

        # 如果名称改变，检查新名称是否已被占用
        new_name = data.get('name', name)
        if new_name != name:
            name_check = session.run(
                "MATCH (kp:KnowledgePoint {name: $name}) RETURN kp",
                name=new_name
            ).single()

            if name_check:
                raise ValueError(f"Knowledge point '{new_name}' already exists")

        # 更新节点
        now = datetime.datetime.utcnow().isoformat()
        session.run("""
            MATCH (kp:KnowledgePoint {name: $old_name})
            SET kp.name = $name,
                kp.category = $category,
                kp.description = $description,
                kp.difficulty = $difficulty,
                kp.importance = $importance,
                kp.estimated_duration = $estimated_duration,
                kp.tags = $tags,
                kp.content = $content,
                kp.updated_at = $updated_at
        """,
            old_name=name,
            name=new_name,
            category=data.get('category'),
            description=data.get('description'),
            difficulty=data.get('difficulty', 'beginner'),
            importance=data.get('importance', 'medium'),
            estimated_duration=data.get('estimated_duration'),
            tags=data.get('tags', []),
            content=data.get('content'),
            updated_at=now
        )

        return get_knowledge_point(new_name)


def delete_knowledge_point(name: str) -> dict:
    """
    删除知识点

    Args:
        name: 知识点名称

    Returns:
        {'success': True, 'message': str}
    """
    with driver.session() as session:
        # 检查是否存在
        existing = session.run(
            "MATCH (kp:KnowledgePoint {name: $name}) RETURN kp",
            name=name
        ).single()

        if not existing:
            raise GraphEntityNotFoundError(f"Knowledge point '{name}' not found")

        # 删除节点及所有关系
        session.run("""
            MATCH (kp:KnowledgePoint {name: $name})
            DETACH DELETE kp
        """, name=name)

        return {
            'success': True,
            'message': f"Knowledge point '{name}' deleted successfully"
        }


def add_prerequisite(knowledge_point_name: str, prerequisite_name: str) -> dict:
    """
    添加前置依赖关系

    Args:
        knowledge_point_name: 知识点名称
        prerequisite_name: 前置依赖的知识点名称

    Returns:
        {'success': True, 'message': str}
    """
    with driver.session() as session:
        # 检查两个节点是否都存在
        result = session.run("""
            MATCH (kp:KnowledgePoint {name: $kp_name})
            MATCH (prereq:KnowledgePoint {name: $prereq_name})
            RETURN kp, prereq
        """,
            kp_name=knowledge_point_name,
            prereq_name=prerequisite_name
        ).single()

        if not result:
            raise GraphEntityNotFoundError("One or both knowledge points not found")

        # 创建REQUIRES关系（kp需要prereq）
        session.run("""
            MATCH (kp:KnowledgePoint {name: $kp_name})
            MATCH (prereq:KnowledgePoint {name: $prereq_name})
            MERGE (kp)-[:REQUIRES]->(prereq)
        """,
            kp_name=knowledge_point_name,
            prereq_name=prerequisite_name
        )

        return {
            'success': True,
            'message': f"Added prerequisite '{prerequisite_name}' to '{knowledge_point_name}'"
        }


def add_relation(
    knowledge_point_name: str,
    related_name: str,
    relationship_type: str = 'RELATED_TO'
) -> dict:
    """
    添加关联关系

    Args:
        knowledge_point_name: 知识点名称
        related_name: 关联的知识点名称
        relationship_type: 关系类型（RELATED_TO/SIMILAR_TO）

    Returns:
        {'success': True, 'message': str}
    """
    if relationship_type not in ['RELATED_TO', 'SIMILAR_TO']:
        raise ValueError("relationship_type must be 'RELATED_TO' or 'SIMILAR_TO'")

    with driver.session() as session:
        # 检查两个节点是否都存在
        result = session.run("""
            MATCH (kp1:KnowledgePoint {name: $name1})
            MATCH (kp2:KnowledgePoint {name: $name2})
            RETURN kp1, kp2
        """,
            name1=knowledge_point_name,
            name2=related_name
        ).single()

        if not result:
            raise GraphEntityNotFoundError("One or both knowledge points not found")

        # 创建关系（双向）
        query = f"""
            MATCH (kp1:KnowledgePoint {{name: $name1}})
            MATCH (kp2:KnowledgePoint {{name: $name2}})
            MERGE (kp1)-[:{relationship_type}]-(kp2)
        """

        session.run(query, name1=knowledge_point_name, name2=related_name)

        return {
            'success': True,
            'message': f"Added {relationship_type} relation between '{knowledge_point_name}' and '{related_name}'"
        }


def get_categories() -> List[str]:
    """
    获取所有知识分类

    Returns:
        List of unique category names
    """
    with driver.session() as session:
        result = session.run("""
            MATCH (kp:KnowledgePoint)
            WHERE kp.category IS NOT NULL
            RETURN DISTINCT kp.category as category
            ORDER BY category
        """)

        return [record['category'] for record in result]


def get_category_tree() -> List[dict]:
    """
    获取分类树结构

    Returns:
        List of categories with knowledge point counts
    """
    with driver.session() as session:
        result = session.run("""
            MATCH (kp:KnowledgePoint)
            WHERE kp.category IS NOT NULL
            WITH kp.category as category, count(kp) as count
            RETURN category, count
            ORDER BY category
        """)

        return [
            {
                'name': record['category'],
                'count': record['count']
            }
            for record in result
        ]

def import_knowledge_points_from_excel(file_content: bytes) -> dict:
    """
    从Excel导入知识点

    Args:
        file_content: Excel文件内容

    Returns:
        {
            'created': int,
            'updated': int,
            'failed': int,
            'errors': List[dict]
        }
    """
    from openpyxl import load_workbook
    from io import BytesIO

    wb = load_workbook(BytesIO(file_content))
    ws = wb.active

    created = 0
    updated = 0
    failed = 0
    errors = []

    # 跳过表头
    rows = list(ws.iter_rows(min_row=2, values_only=True))

    for idx, row in enumerate(rows, start=2):
        if not row or not row[0]:  # 跳过空行
            continue

        try:
            data = {
                'name': row[0],
                'category': row[1] if len(row) > 1 else None,
                'description': row[2] if len(row) > 2 else None,
                'difficulty': row[3] if len(row) > 3 else 'beginner',
                'importance': row[4] if len(row) > 4 else 'medium',
                'estimated_duration': int(row[5]) if len(row) > 5 and row[5] else None,
                'tags': [t.strip() for t in row[6].split(',')] if len(row) > 6 and row[6] else [],
                'content': row[7] if len(row) > 7 else None
            }

            # 尝试创建，如果存在则更新
            try:
                create_knowledge_point(data)
                created += 1
            except ValueError as e:
                if "already exists" in str(e):
                    update_knowledge_point(data['name'], data)
                    updated += 1
                else:
                    raise

        except Exception as e:
            failed += 1
            errors.append({
                'row': idx,
                'error': str(e)
            })

    return {
        'created': created,
        'updated': updated,
        'failed': failed,
        'errors': errors
    }

    """CSV导入知识点"""
    if 'file' not in request.files:
        return jsonify({"error": "No file provided"}), 400

    file = request.files['file']

    if not file or file.filename == '':
        return jsonify({"error": "No file selected"}), 400

    try:
        content = file.read().decode('utf-8')
        result = graph_service.import_knowledge_points_from_csv(content)
        return jsonify(result), 200
    except Exception as e:
        return jsonify({"error": str(e)}), 500

@require_role("teacher")

        content = graph_service.export_knowledge_points_to_excel()
        filename = f"knowledge_points_{datetime.date.today().isoformat()}.xlsx"

// static/js/admin.js

async function refreshAdminGraph() {
  if (!adminGraphCanvas) return;

  try {
    if (adminGraphStatus) {
      adminGraphStatus.textContent = "加载中...";
    }

    // 使用新的enhanced端点
    const [networkResp, knowledgeResp] = await Promise.all([
      fetchWithAuth("/api/graph/network"),
      fetchWithAuth("/api/graph/knowledge-points/enhanced"),  // 改为新端点
    ]);

    if (!networkResp.ok) {
      if (networkResp.status === 503) {
        throw new Error("知识图谱服务暂不可用");
      }
      throw new Error("无法加载知识图谱");
    }

    const networkData = await networkResp.json();
    state.admin.graph.network = networkData || { nodes: [], edges: [] };

    if (knowledgeResp.ok) {
      const knowledgeData = await knowledgeResp.json();
      state.admin.graph.knowledgePoints = knowledgeData.knowledge_points || [];  // 注意字段名
    }

    renderAdminGraphKnowledgeList();
    renderAdminGraphNetwork();

    if (adminGraphStatus) {
      const nodeCount = (networkData.nodes || []).length;
      const edgeCount = (networkData.edges || []).length;
      adminGraphStatus.textContent = `节点 ${nodeCount} · 关系 ${edgeCount}`;
    }
  } catch (error) {
    console.error(error);
    if (adminGraphStatus) {
      adminGraphStatus.textContent = error.message || "加载知识图谱失败";
    }
  }
}
